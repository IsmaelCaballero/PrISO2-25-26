    private record Fila(Map<String, String> celdas) {

        String get(String columna) {
            return celdas.get(columna);
        }

        String id() {
            return get("id").replace("TC14-", "");
        }
    }

    private static List<Fila> leer(String nombre) {
        List<Path> candidatos = List.of(
                Paths.get("iso2-citas", "src", "test", "resources", "tdop", nombre),
                Paths.get("src", "test", "resources", "tdop", nombre));
        Path path = candidatos.stream().filter(Files::exists).findFirst()
                .orElseThrow(() -> new IllegalStateException("No se encuentra la matriz " + nombre));
        try {
            List<String> lineas = Files.readAllLines(path, StandardCharsets.UTF_8);
            String[] cabecera = lineas.get(0).split(";", -1);
            List<Fila> filas = new ArrayList<>();
            for (String linea : lineas.subList(1, lineas.size())) {
                if (linea.isBlank()) {
                    continue;
                }
                String[] celdas = linea.split(";", -1);
                Map<String, String> fila = new LinkedHashMap<>();
                for (int i = 0; i < cabecera.length; i++) {
                    fila.put(cabecera[i].trim(), i < celdas.length ? celdas[i].trim() : "");
                }
                filas.add(new Fila(fila));
            }
            return filas;
        } catch (IOException e) {
            throw new UncheckedIOException("No se pudo leer la matriz " + nombre, e);
        }
    }

    private static Map<String, String[]> leerEsperados() {
        Map<String, String[]> tabla = new LinkedHashMap<>();
        for (Fila f : leer("cdu14-esperados.csv")) {
            tabla.put(f.get("esperado"),
                    new String[] { f.get("esperadoStatus"), f.get("esperadoCodigo") });
        }
        return tabla;
    }
